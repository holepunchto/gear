1// bare-svelte — minimal Svelte 5 harness running on Bare
2// Replicates over hyperswarm; no dev-server required.
3
4const Corestore = require('corestore')
5const Hyperswarm = require('hyperswarm')
6const { mount } = require('./lib/svelte')
7
8async function main() {
9 const store = new Corestore('./data')
10 const swarm = new Hyperswarm()
11
12 swarm.on('connection', (conn) => {
13 store.replicate(conn)
14 })
15
16 const core = store.get({ name: 'ui' })
17 await core.ready()
18 swarm.join(core.discoveryKey, { server: true, client: true })
19
20 const app = await mount({ target: document.body, props: { core } })
21 return app
22}
23
24main().catch((err) => {
25 console.error(err)
26 process.exit(1)
27})
